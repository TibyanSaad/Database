SELECT FNAME || ' ' || LNAME AS FULL_NAME, DNAME, PNAME 
FROM EMPLOYEES E 
JOIN DEPARTMENTS D ON E.DNUM = D.DNUM
JOIN PROJECTS P ON D.DNUM=P.DNUM;


SELECT FNAME || ' ' || LNAME AS FULL_NAME, D.*,P.*
FROM EMPLOYEES E
LEFT JOIN DEPARTMENTS D ON D.DNUM=E.DNUM
LEFT JOIN DEPENDENTS P ON E.SSN=P.SSN;


SELECT PNAME,DNAME, FNAME || ' ' || LNAME AS FULL_NAME
FROM EMPLOYEES E 
RIGHT JOIN DEPARTMENTS D ON D.DNUM=E.DNUM
RIGHT JOIN PROJECTS P ON P.DNUM=D.DNUM;

SELECT E.FNAME || ' ' || E.LNAME AS EMPLOYEE_NAME, M.FNAME || ' ' || M.LNAME AS SUPERVISOR_NAME, D.DNAME
FROM EMPLOYEES E
JOIN EMPLOYEES M ON M.SSN = E.SUPERVISOR_ID
JOIN DEPARTMENTS D ON D.DNUM =E.DNUM;
